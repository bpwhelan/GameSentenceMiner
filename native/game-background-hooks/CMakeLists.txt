cmake_minimum_required(VERSION 3.24)
project(gsm-game-background LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Game background hooks require Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
add_library(minhook STATIC
    third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c
    third_party/minhook/src/hde/hde32.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(gsm-game-background SHARED hooks.cpp)
target_link_libraries(gsm-game-background PRIVATE minhook user32)
add_executable(gsm-game-background-helper helper.cpp)
target_link_libraries(gsm-game-background-helper PRIVATE user32 version)
enable_testing()
add_executable(game-background-policy-test tests/policy_test.cpp)
target_include_directories(game-background-policy-test PRIVATE .)
add_test(NAME policy COMMAND game-background-policy-test)
add_library(game-background-fake-xinput SHARED tests/fake_xinput.cpp tests/fake_xinput.def)
set_target_properties(game-background-fake-xinput PROPERTIES OUTPUT_NAME xinput1_3)
add_executable(game-background-fixture tests/fixture.cpp)
target_link_libraries(game-background-fixture PRIVATE user32)
